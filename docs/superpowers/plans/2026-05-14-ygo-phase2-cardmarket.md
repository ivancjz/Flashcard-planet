> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ingest CardMarket avg7/avg30/avg1 prices daily for YGO assets, route them through a CardMarket-specific signal compute path with a dispersion quality gate, and surface source context in Discord alerts.

**Architecture:** Three new `source` values (`cardmarket_avg7`, `cardmarket_avg30`, `cardmarket_avg1`) are written to `price_history` daily; the signal engine detects these sources and computes delta as `avg7 / avg30` with a dispersion gate instead of the existing dual-window algorithm; Discord heartbeat gains per-game signal breakdown.

**Tech Stack:** Python 3.13, SQLAlchemy 2, APScheduler, httpx, SQLite (tests), PostgreSQL 18 (prod). CardMarket S3 public JSON files (no auth).

The catalog maps CardMarket `idProduct` → card name. We download it once per ingest run and build a `name → [idProduct]` lookup. The catalog is ~14 MB. Files are regenerated by CardMarket ~02:43 CET daily; an anomalous scheduler restart could trigger a redundant 28 MB fetch (catalog + price guide). To prevent this, the downloader supports **ETag-based conditional fetch**: pass the last-seen ETag and receive `None` back on HTTP 304 Not Modified, allowing the ingest caller to skip cleanly. The ETag is persisted in `scheduler_run_log.meta_json` between runs so it survives Railway deploys.

"""CardMarket public S3 catalog downloader.

Downloads the daily price guide + singles catalog for a given game ID.
No authentication required. Files are regenerated by CardMarket ~02:43 CET daily.

Usage:
    cat = CardmarketCatalog.download(game_id=GAME_ID_YGO)
    ids = cat.product_ids_for_name("Ash Blossom & Joyous Spring")
    prices = cat.prices_for_product(101788)  # {"avg7": 5.0, "avg30": 4.5, "avg1": 6.0}
"""
from __future__ import annotations

import logging
from dataclasses import dataclass, field

import httpx

logger = logging.getLogger(__name__)

GAME_ID_YGO = 3
GAME_ID_POKEMON = 6
GAME_ID_MTG = 1

_S3_BASE = "https://downloads.s3.cardmarket.com/productCatalog"
CM_SINGLES_CATALOG_URL = _S3_BASE + "/productList/products_singles_{game_id}.json"
CM_PRICE_GUIDE_URL = _S3_BASE + "/priceGuide/price_guide_{game_id}.json"

_TIMEOUT = 60.0  # seconds; files are 10–25 MB


@dataclass
class CardmarketCatalog:
    game_id: int
    created_at: str
    etag: str = ""                # ETag from the price guide response; store in run meta_json
    # name (lowercased) → list of idProduct
    _name_to_ids: dict[str, list[int]] = field(default_factory=dict, repr=False)
    # idProduct → {avg7, avg30, avg1} (float | None)
    _prices: dict[int, dict[str, float | None]] = field(default_factory=dict, repr=False)

    @classmethod
    def download(
        cls,
        *,
        game_id: int = GAME_ID_YGO,
        etag: str | None = None,
    ) -> "CardmarketCatalog | None":
        """Download catalog + price guide from CardMarket S3. No auth required.

        Pass etag (from a previous run's meta_json) to enable conditional fetch.
        Returns None on HTTP 304 Not Modified — caller should skip ingest entirely.
        """
        catalog_url = CM_SINGLES_CATALOG_URL.format(game_id=game_id)
        price_url = CM_PRICE_GUIDE_URL.format(game_id=game_id)

        # Condition the price guide fetch on ETag — it's the file that changes daily.
        # The catalog (product list) changes less frequently; we always re-fetch it
        # because it's needed to resolve new cards added since the last run.
        price_headers = {"If-None-Match": etag} if etag else {}

        logger.info("cardmarket_catalog_download game_id=%s etag=%s", game_id, etag or "none")

        price_resp = httpx.get(price_url, headers=price_headers, timeout=_TIMEOUT)
        if price_resp.status_code == 304:
            logger.info("cardmarket_price_guide_not_modified game_id=%s", game_id)
            return None  # Caller skips ingest
        price_resp.raise_for_status()
        price_data = price_resp.json()
        new_etag = price_resp.headers.get("ETag", "")

        catalog_resp = httpx.get(catalog_url, timeout=_TIMEOUT)
        catalog_resp.raise_for_status()
        catalog_data = catalog_resp.json()

        name_to_ids: dict[str, list[int]] = {}
        for product in catalog_data.get("products", []):
            name_key = product["name"].lower().strip()
            name_to_ids.setdefault(name_key, []).append(product["idProduct"])

        prices: dict[int, dict[str, float | None]] = {}
        for entry in price_data.get("priceGuides", []):
            pid = entry["idProduct"]
            prices[pid] = {
                "avg1":  entry.get("avg1"),
                "avg7":  entry.get("avg7"),
                "avg30": entry.get("avg30"),
            }

        logger.info(
            "cardmarket_catalog_loaded game_id=%s products=%s priced=%s etag=%s",
            game_id, len(name_to_ids), len(prices), new_etag,
        )
        return cls(
            game_id=game_id,
            created_at=catalog_data.get("createdAt", ""),
            etag=new_etag,
            _name_to_ids=name_to_ids,
            _prices=prices,
        )

    def product_ids_for_name(self, name: str) -> list[int]:
        """Return all CardMarket product IDs matching this card name (case-insensitive)."""
        return list(self._name_to_ids.get(name.lower().strip(), []))

    def prices_for_product(self, product_id: int) -> dict[str, float | None]:
        """Return {avg1, avg7, avg30} for a product ID. Returns all-None if unknown."""
        return self._prices.get(product_id, {"avg1": None, "avg7": None, "avg30": None})
```

**Matching algorithm (MVP):**
1. Check `asset.metadata_json.get('cm_product_ids', [])` for a pre-cached list.
2. If empty, look up by `asset.name` in the catalog. Cache the result.
3. If a card has multiple CM products (multiple printings), write price rows for each, tagged with the product ID in a note. For Phase 2, this is acceptable — we're aggregating across printings.

**Source naming:**
- `cardmarket_avg7` — 7-day average of completed sales (primary current signal)
- `cardmarket_avg30` — 30-day average of completed sales (baseline)
- `cardmarket_avg1` — 1-day average (dispersion proxy only; optional — skip if null)

"""CardMarket daily price ingest for YGO assets.

Writes up to 3 price_history rows per asset per run:
  source='cardmarket_avg7'  — 7-day avg of completed sales (primary signal)
  source='cardmarket_avg30' — 30-day avg of completed sales (baseline)
  source='cardmarket_avg1'  — 1-day avg (dispersion proxy; skipped when null)

All rows: currency='EUR', market_segment='raw'.
Cards without avg7 AND avg30 data are skipped.
"""
from __future__ import annotations

import logging
import uuid
from dataclasses import dataclass, field
from datetime import UTC, datetime
from decimal import Decimal

from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.app.ingestion.game_data.cardmarket_catalog import CardmarketCatalog, GAME_ID_YGO
from backend.app.models.asset import Asset
from backend.app.models.price_history import PriceHistory

logger = logging.getLogger(__name__)

CM_SOURCE_AVG1 = "cardmarket_avg1"
CM_SOURCE_AVG7 = "cardmarket_avg7"
CM_SOURCE_AVG30 = "cardmarket_avg30"


@dataclass
class CardmarketIngestionResult:
    assets_matched: int = 0
    assets_skipped_no_match: int = 0
    assets_skipped_no_price: int = 0
    price_points_written: int = 0
    captured_at: datetime | None = None
    skipped_not_modified: bool = False   # True when S3 returned 304
    catalog_etag: str = ""               # Store back in scheduler meta_json


def ingest_cardmarket_ygo(
    session: Session,
    *,
    catalog: CardmarketCatalog | None = None,
    last_etag: str | None = None,
) -> CardmarketIngestionResult:
    """Match YGO assets to CardMarket products and write avg1/avg7/avg30 to price_history.

    If catalog is None, downloads fresh from S3.
    Pass last_etag (from scheduler_run_log.meta_json) to skip on 304.
    """
    if catalog is None:
        downloaded = CardmarketCatalog.download(game_id=GAME_ID_YGO, etag=last_etag)
        if downloaded is None:
            # S3 returned 304 — files unchanged since last run, skip ingest
            result = CardmarketIngestionResult()
            result.skipped_not_modified = True
            return result
        catalog = downloaded

    result = CardmarketIngestionResult()
    result.captured_at = datetime.now(UTC).replace(microsecond=0)

    # Fetch all YGO assets from DB
    ygo_assets = session.scalars(
        select(Asset).where(Asset.game == "yugioh")
    ).all()

    for asset in ygo_assets:
        # Resolve product IDs: cached first, then name lookup
        cached_ids: list[int] = (asset.metadata_json or {}).get("cm_product_ids", [])
        if not cached_ids:
            cached_ids = catalog.product_ids_for_name(asset.name)
            if cached_ids:
                meta = dict(asset.metadata_json or {})
                meta["cm_product_ids"] = cached_ids
                asset.metadata_json = meta
                session.flush()

        if not cached_ids:
            result.assets_skipped_no_match += 1
            logger.debug("cardmarket_no_match asset=%s name=%r", asset.id, asset.name)
            continue

        # Write price rows for each matched product
        matched = False
        for pid in cached_ids:
            prices = catalog.prices_for_product(pid)
            avg7 = prices.get("avg7")
            avg30 = prices.get("avg30")
            avg1 = prices.get("avg1")

            if avg7 is None and avg30 is None:
                continue  # This printing has no price data

            matched = True
            rows_to_write: list[tuple[str, float]] = []
            if avg7 is not None:
                rows_to_write.append((CM_SOURCE_AVG7, avg7))
            if avg30 is not None:
                rows_to_write.append((CM_SOURCE_AVG30, avg30))
            if avg1 is not None:
                rows_to_write.append((CM_SOURCE_AVG1, avg1))

            for source, price_val in rows_to_write:
                session.add(PriceHistory(
                    id=uuid.uuid4(),
                    asset_id=asset.id,
                    source=source,
                    currency="EUR",
                    price=Decimal(str(price_val)),
                    captured_at=result.captured_at,
                    market_segment="raw",
                ))
                result.price_points_written += 1

        if matched:
            result.assets_matched += 1
        else:
            result.assets_skipped_no_price += 1

    result.catalog_etag = catalog.etag
    logger.info(
        "cardmarket_ingest_complete matched=%s no_match=%s no_price=%s written=%s etag=%s",
        result.assets_matched, result.assets_skipped_no_match,
        result.assets_skipped_no_price, result.price_points_written, catalog.etag,
    )
    return result
```

```python
def compute_cardmarket_delta(
    db: Session,
    asset_id: Any,
    *,
    now: datetime | None = None,
) -> tuple[Decimal | None, dict]:
    """Compute price delta for an asset using CardMarket avg7/avg30 sources.

    current_price  = most recent cardmarket_avg7 within CM_CURRENT_WINDOW_HOURS
    baseline_price = most recent cardmarket_avg30 within CM_CURRENT_WINDOW_HOURS
    dispersion     = |avg1 - avg7| / avg7 — gate for illiquid/noisy data

    Returns (delta_pct, ctx). delta_pct is None on insufficient data.
    """
    if now is None:
        now = datetime.now(UTC)
    window_start = now - timedelta(hours=CM_CURRENT_WINDOW_HOURS)

    rows = db.execute(
        select(PriceHistory.source, PriceHistory.price)
        .where(
            PriceHistory.asset_id == asset_id,
            PriceHistory.source.in_(CM_SOURCES),
            PriceHistory.captured_at >= window_start,
            PriceHistory.captured_at <= now,
            PriceHistory.market_segment == "raw",
        )
        .order_by(PriceHistory.captured_at.desc())
    ).all()

    # Take the most recent row for each source type
    by_source: dict[str, Decimal] = {}
    for row in rows:
        if row.source not in by_source:
            by_source[row.source] = Decimal(str(row.price))

    avg7 = by_source.get(CM_SOURCE_AVG7)
    avg30 = by_source.get(CM_SOURCE_AVG30)
    avg1 = by_source.get(CM_SOURCE_AVG1)

    if avg7 is None and avg30 is None:
        return None, {"reason": "no_cardmarket_data"}

    if avg30 is None:
        return None, {"reason": "no_baseline", "avg7": float(avg7)}

    if avg7 is None:
        return None, {"reason": "no_current", "avg30": float(avg30)}

    # Bulk floor gate (same threshold as standard path)
    if avg30 < SIGNAL_BULK_FLOOR_PRICE:
        return None, {"reason": "bulk_baseline_price", "avg30": float(avg30)}

    # Dispersion gate: large avg1 spike → unreliable signal
    if avg1 is not None and avg7 > 0:
        dispersion_ratio = abs(avg1 - avg7) / avg7
        if dispersion_ratio > CARDMARKET_DISPERSION_THRESHOLD:
            return None, {
                "reason": "high_dispersion",
                "avg1": float(avg1),
                "avg7": float(avg7),
                "dispersion_ratio": float(dispersion_ratio),
            }

    if avg30 == 0:
        return None, {"reason": "zero_baseline"}

    delta = ((avg7 - avg30) / avg30 * Decimal("100")).quantize(Decimal("0.01"))
    return delta, {
        "current_price": float(avg7),
        "baseline_price": float(avg30),
        "avg1": float(avg1) if avg1 is not None else None,
        "source": "cardmarket",
    }

def _get_signal_counts_by_game(db: Session) -> dict[str, dict[str, int]]:
    """Return {game: {label: count}} from asset_signals joined to assets."""
    from backend.app.models.asset_signal import AssetSignal
    from backend.app.models.asset import Asset
    rows = db.execute(
        select(Asset.game, AssetSignal.label, func.count().label("cnt"))
        .join(Asset, Asset.id == AssetSignal.asset_id)
        .group_by(Asset.game, AssetSignal.label)
    ).all()
    result: dict[str, dict[str, int]] = {}
    for row in rows:
        result.setdefault(row.game, {})[row.label] = row.cnt
    return result
```
